Analysis II: Bayesian multilevel models on Principal Components (1-5)

Overview

In the previous script (4_PCA_featureDimensions), we used PCA to re-structure multidimensional kinematic space into fewer components. PCA is unsupervised which means that it does not know which datapoints belong to which group.

In this script, we fit one model per component with neutral cell-means coding (0 + groupID) and pull every comparison of interest after the fit with emmeans.

The main comparisons include (all in SD units, net of speaker & item):

  • gesture vs native sign — conventionalization, and more broadly, gesture_vs_all_signers.
  • learner1 – learner2 – signer progression — mainly mean(L1, L2) vs native signers, additionaly also naive_vs_beginner (L1 vs L2, within-subject), beginner_vs_native (L2 vs Sign), and mean(L1)

To make the process efficient, we build functions that will allow us to fit all the models in a loop. Essentially, each model takes a component of one PCA (pooled-all, pooled-gross, pooled-fine, sign-only, sign-gross, sign-fine) and then saves the output such that it is recognizable which model the component/output belongs to.

# read one domain's score file, prefix component columns with the domain ---
read_scores <- function(basis, domain) {
  suf  <- switch(basis, varimax = "_varimax", signonly = "_signonly", "")
  path <- file.path("Datasets", sprintf("pca_scores_%s%s.csv", domain, suf))
  if (!file.exists(path)) return(NULL)
  read_csv(path, show_col_types = FALSE) %>%
    rename_with(~ paste0(domain, "_", .x), matches("^(PC|RC)[0-9]+$"))
}

# build the modelling frame for a basis: merge all/gross/fine, harmonize concepts, factor, z-score 
build_d <- function(basis) {
  blocks <- map(c("all", "gross", "fine"), ~ read_scores(basis, .x))
  if (any(map_lgl(blocks, is.null))) return(NULL)   # a domain file missing -> skip basis
  names(blocks) <- c("all", "gross", "fine")

  comp_cols <- unlist(map(names(blocks), function(dm)
    grep(sprintf("^%s_(PC|RC)[0-9]+$", dm), names(blocks[[dm]]), value = TRUE)))

  key <- c("participant", "groupID", "concept")
  d <- reduce(blocks, full_join, by = key)

  # harmonize concept labels: native-signer English glosses -> German labels 
  # (done after the join, because the all/gross/fine score files share the original labels)
  concept_map <- read_csv(file.path("Datasets", "concept_map.csv"), show_col_types = FALSE)
  n_before <- n_distinct(d$concept)
  d <- d %>%
    left_join(select(concept_map, concept, concept_id), by = "concept") %>%
    mutate(concept = coalesce(concept_id, concept)) %>%
    select(-concept_id)
  message("[", basis, "] concepts: ", n_before, " -> ", n_distinct(d$concept),
          " after harmonization")

  d <- d %>%
    mutate(groupID     = factor(groupID, levels = group_levels),
           participant = factor(participant),
           concept     = factor(concept)) %>%
    mutate(across(all_of(comp_cols), ~ as.numeric(scale(.)), .names = "{.col}_z"))

  # NA report from the full joins (tokens not shared across all/gross/fine)
  na_rows <- d %>% summarise(across(all_of(comp_cols), ~ sum(is.na(.)))) %>%
    pivot_longer(everything(), names_to = "component", values_to = "n_NA") %>%
    filter(n_NA > 0)
  if (nrow(na_rows)) { message("NA scores after join (", basis, "):"); print(na_rows) }

  stopifnot(all(levels(d$groupID) == group_levels))
  list(d = d, comp_cols = comp_cols)
}

# fit gaussian model for one component, cached per basis+component
fit_component <- function(comp, d, basis) {
  form <- bf(as.formula(paste0(comp, "_z ~ 0 + groupID + (1|participant) + (1|concept)")))
  brm(form, data = d, prior = pr, family = gaussian(),
      chains = 4, iter = 8000, warmup = 4000, seed = 1,
      control = list(adapt_delta = 0.95),
      file = file.path("models", paste0("brm_", basis, "_", comp)),
      file_refit = "on_change")
}

# extract all comparisons for one fit via emmeans (population-level: re_formula = NA)
extract_comparisons <- function(fit, comp, basis) {
  emm <- emmeans(fit, ~ groupID, re_formula = NA)     # 4 marginal means, net of RE
  w   <- if (basis == "signonly") comparisons_sign else comparisons   # 3- vs 4-group weights
  con <- contrast(emm, method = w)                    # weights positional in group order
  gather_emmeans_draws(con) %>%
    group_by(contrast) %>%
    summarise(estimate = median(.value),
              .lower   = quantile(.value, 0.025),
              .upper   = quantile(.value, 0.975),
              pd       = max(mean(.value > 0), mean(.value < 0)),
              .groups  = "drop") %>%
    mutate(component = comp,
           basis     = basis,
           domain    = str_extract(comp, "^(all|gross|fine)"),
           credible  = sign(.lower) == sign(.upper))
}

# convergence diagnostics for one fit 
diagnose <- function(fit) {
  np  <- brms::nuts_params(fit)
  drw <- as_draws_df(fit)
  ess <- summarise_draws(drw, "ess_bulk", "ess_tail")
  tibble(max_rhat     = max(brms::rhat(fit), na.rm = TRUE),
         min_ess_bulk = min(ess$ess_bulk, na.rm = TRUE),
         min_ess_tail = min(ess$ess_tail, na.rm = TRUE),
         n_divergent  = sum(np$Value[np$Parameter == "divergent__"]))
}

# read + tidy loadings for a basis (PC or RC id column -> prefixed `component`) 
read_loadings <- function(basis, domain) {
  suf  <- switch(basis, varimax = "_varimax", signonly = "_signonly", "")
  path <- file.path("Datasets", sprintf("loadings_%s%s.csv", domain, suf))
  if (!file.exists(path)) return(NULL)
  x  <- read_csv(path, show_col_types = FALSE)
  id <- if (basis == "varimax") "RC" else "PC"
  if (!"sig_loading" %in% names(x) && "CI_above_null_after" %in% names(x))
    x <- rename(x, sig_loading = CI_above_null_after)     # tolerate the older column name
  x %>%
    rename(comp_id = all_of(id)) %>%
    mutate(comp_id   = as.character(comp_id),
           comp_id   = if_else(str_detect(comp_id, "^[0-9]+$"), paste0(id, comp_id), comp_id),
           component = paste0(domain, "_", comp_id),
           basis     = basis)
}

Here we define the functions that will allow us to run the models in a loop, or load them if already saved

run_basis <- function(basis) {
  bd <- build_d(basis)
  if (is.null(bd)) { message("Skipping basis '", basis, "': score files missing."); return(NULL) }
  d <- bd$d; comp_cols <- bd$comp_cols
  message("[", basis, "] modelling: ", paste(comp_cols, collapse = ", "))

  fits <- setNames(lapply(comp_cols, fit_component, d = d, basis = basis), comp_cols)

  list(
    basis     = basis,
    d         = d,
    fits      = fits,
    comp_cols = comp_cols,
    coefs     = map_dfr(comp_cols, ~ extract_comparisons(fits[[.x]], .x, basis)),
    converge  = imap_dfr(fits, ~ mutate(diagnose(.x), component = .y, basis = basis)),
    n_part    = d %>% distinct(groupID, participant) %>% count(groupID, name = "n_participants")
  )
}

load_or_run <- function(basis, cache_dir = "models", refit = FALSE) {
  f <- file.path(cache_dir, paste0("run_", basis, ".rds"))
  if (!refit && file.exists(f)) {
    message("[", basis, "] loading cached run")
    return(readRDS(f))
  }
  res <- run_basis(basis)
  if (!is.null(res)) {
    dir.create(cache_dir, showWarnings = FALSE, recursive = TRUE)
    tmp <- tempfile(tmpdir = cache_dir, fileext = ".rds.part")
    ok <- tryCatch({
      saveRDS(res, tmp, compress = "xz")   # smaller files; use FALSE if speed matters more
      file.rename(tmp, f)
    }, error = function(e) {
      unlink(tmp)
      warning("[", basis, "] caching failed: ", conditionMessage(e))
      FALSE
    })
  }
  res
}

Interpretation — which features drive each credible comparison

Join credible comparisons to the (basis-matched) significant loadings. Direction: a positive comparison estimate on a positively-loading feature => the first group shows more of it.

Display code 
loadings <- map_dfr(
  c("pca"),
  function(b) map_dfr(c("all", "gross", "fine"), ~ read_loadings(b, .x))
)

interpretation <- coefs %>%
  filter(credible) %>%
  select(basis, component, contrast, effect = estimate) %>%
  inner_join(loadings, by = c("basis", "component")) %>%
  filter(sig_loading) %>%
  mutate(
    # signed loading (works whether obs_index is stored signed or unsigned)
    loading  = abs(obs_index) * loading_sign,
    # split leading category from the measure name
    category = if_else(str_detect(variable, "^[^_]+__[^_]+__"),
                       str_extract(variable, "^[^_]+__[^_]+"),   # e.g. "gross__rwrist"
                       NA_character_),
    measure  = if_else(str_detect(variable, "^[^_]+__[^_]+__"),
                       str_remove(variable, "^[^_]+__[^_]+__"),  # e.g. "speed_mean"
                       variable),
    moves    = if_else(sign(effect) == sign(loading), "↑ more", "↓ less")
  ) %>%
  arrange(basis, contrast, component, category, desc(abs(loading))) %>%
  relocate(category, measure, loading, .after = component)

dim_table <- function(spec, title, file, notes_extra = NULL) {
  # spec: tibble(basis, domain, block); one row group per spec row, in that order
  spec <- spec %>%
    mutate(block = paste0(block, " (", map2_chr(basis, domain, n_info), ")"))
  keys <- select(spec, basis, domain)

  var_tab <- var_all %>%
    semi_join(keys, by = c("basis", "domain")) %>%
    transmute(basis, domain, component = paste0(domain, "_PC", PC),
              variance = sprintf("%.1f%%<br>[%.1f, %.1f]<br><small>cum. %.0f%%</small>",
                                 100 * var_exp, 100 * var_low, 100 * var_high, 100 * cum_var))

  # rows = every modeled component of every PCA (not only those in the variance file)
  rows <- bind_rows(
    coefs_all %>% semi_join(keys, by = c("basis", "domain")) %>% distinct(basis, domain, component),
    select(var_tab, basis, domain, component)
  ) %>% distinct()

  missing_var <- anti_join(keys, distinct(var_tab, basis, domain), by = c("basis", "domain"))
  if (nrow(missing_var) > 0)
    warning("No variance rows for: ", paste(missing_var$basis, missing_var$domain, sep = "/", collapse = ", "),
            ". Check the `domain` values in pca_variance_explained.csv: ",
            paste(unique(var_all$domain), collapse = ", "))

  feat_tab <- pmap_dfr(keys, read_loadings) %>%
    add_count(basis, component, name = "n_sig") %>%
    filter(within_50pct) %>%
    arrange(basis, component, desc(share)) %>%
    group_by(basis, component, n_sig) %>%
    summarise(n_top = n(),
              features = paste0(if_else(loading_sign > 0, "&#43;", "−"), " ", pretty_feat(variable),
                                " (", round(100 * share), "%)", collapse = "<br>"),
              .groups = "drop") %>%
    mutate(features = paste0(features, "<br><small><i>", n_top, " of ", n_sig,
                             " significant features</i></small>")) %>%
    select(basis, component, features)

  con_tab <- coefs_all %>%
    semi_join(keys, by = c("basis", "domain")) %>%
    mutate(cell = sprintf("%s%.2f<br>[%.2f, %.2f]%s<br><small>pd = %.3f</small>",
                          if_else(credible, "<b>", ""), estimate, .lower, .upper,
                          if_else(credible, "</b>", ""), pd)) %>%
    select(basis, component, contrast, cell) %>%
    pivot_wider(names_from = contrast, values_from = cell)

  conv_tab <- conv_all %>%
    semi_join(keys, by = c("basis", "domain")) %>%
    transmute(basis, component,
              diag = sprintf("%.3f<br>%.0f<br>%d", max_rhat, min_ess_bulk, as.integer(n_divergent)))

  tab <- rows %>%
    left_join(select(var_tab, -domain), by = c("basis", "component")) %>%
    left_join(pc_labels, by = c("basis", "component")) %>%
    left_join(feat_tab,  by = c("basis", "component")) %>%
    left_join(con_tab,   by = c("basis", "component")) %>%
    left_join(conv_tab,  by = c("basis", "component")) %>%
    left_join(select(spec, basis, domain, block), by = c("basis", "domain")) %>%
    mutate(Dim = dim_of(component), block = factor(block, levels = spec$block)) %>%
    arrange(block, parse_number(Dim))

  # keep only contrasts that exist in at least one block of this table
  contr <- contr_all[contr_all %in% names(tab)]
  contr <- contr[map_lgl(contr, ~ any(!is.na(tab[[.x]])))]

  gt_tab <- tab %>%
    select(block, Dim, label, variance, features, all_of(contr), diag) %>%
    gt(groupname_col = "block") %>%
    tab_header(title = title) %>%
    cols_label(.list = c(list(Dim = "Dim.", label = "Interpretation",
                              variance = html("Variance<br>explained"),
                              features = html("Features up to 50%<br>of index loading"),
                              diag     = html("R̂ / ESS /<br>div.")),
                         contr_lab[contr])) %>%
    tab_spanner("Dimension", columns = c(Dim, label, variance, features)) %>%
    tab_spanner("Group comparisons: estimate [95% CrI], SD units", columns = all_of(contr)) %>%
    fmt_markdown(columns = c(variance, features, all_of(contr), diag)) %>%
    sub_missing(missing_text = "—") %>%
    cols_width(features ~ px(260), label ~ px(140)) %>%
    tab_style(style = cell_text(size = px(11)), locations = cells_body()) %>%
    tab_style(style = cell_text(weight = "bold"), locations = cells_row_groups())

  for (n in c(notes_extra, notes_base)) gt_tab <- tab_source_note(gt_tab, md(n))
  gtsave(gt_tab, file)
  gt_tab
}

# ===================================================================
# 2. Compact summary: which domain × dimension carries which trajectory
# ===================================================================
traj_map <- function(bases, title, file) {
  present <- coefs_all %>%
    filter(basis %in% bases) %>%
    distinct(basis, domain, contrast)

  cred <- coefs_all %>%
    filter(basis %in% bases, credible) %>%
    left_join(pc_labels, by = c("basis", "component")) %>%
    mutate(cell = sprintf("%s %+.2f%s", dim_of(component), estimate,
                          if_else(is.na(label), "", paste0(" <small>", label, "</small>")))) %>%
    group_by(basis, domain, contrast) %>%
    summarise(cell = paste(cell, collapse = "<br>"), .groups = "drop")

  tab <- present %>%
    left_join(cred, by = c("basis", "domain", "contrast")) %>%
    mutate(cell = coalesce(cell, "—")) %>%                         # estimable, nothing credible
    pivot_wider(names_from = contrast, values_from = cell) %>%    # NA = not estimable
    left_join(select(basis_info, basis, title), by = "basis") %>%
    mutate(block  = factor(title, levels = basis_info$title[basis_info$basis %in% bases]),
           domain = factor(domain, levels = dom_levels, labels = dom_lab[dom_levels])) %>%
    arrange(block, domain)

  contr <- contr_all[contr_all %in% names(tab)]

  gt_tab <- tab %>%
    select(block, domain, all_of(contr)) %>%
    gt(groupname_col = "block") %>%
    tab_header(title = title) %>%
    cols_label(.list = c(list(domain = "Domain"), contr_lab[contr])) %>%
    fmt_markdown(columns = all_of(contr)) %>%
    sub_missing(missing_text = "n/a") %>%
    tab_style(style = cell_text(size = px(11)), locations = cells_body()) %>%
    tab_style(style = cell_text(weight = "bold"), locations = cells_row_groups()) %>%
    tab_source_note(md(paste(
      "Cells list the dimensions whose 95% CrI for that contrast excludes 0, with the posterior mean (SD units).",
      "— = estimable but no credible difference; n/a = contrast not estimable in that PCA (no gesturers).",
      "Dimension numbers refer to each domain's own PCA; see the detailed tables for intervals and loadings.")))

  gtsave(gt_tab, file)
  gt_tab
}
